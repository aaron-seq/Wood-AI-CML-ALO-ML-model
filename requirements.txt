# Runtime dependencies for the API and the ML pipeline.
#
# Versions carry upper bounds on purpose: the committed model artifact in
# models/ is a scikit-learn pickle, and unpickling it under a different
# minor version of scikit-learn emits an InconsistentVersionWarning and is
# not guaranteed to reproduce the same predictions. Raising the
# scikit-learn bound therefore requires retraining and re-committing the
# model -- see docs/DEPLOYMENT.md, "Upgrading scikit-learn".
#
# The dashboard's extra dependencies live in requirements-streamlit.txt so
# the API container does not have to ship a UI stack.

# API server.
fastapi>=0.128,<0.142
# starlette is a transitive dependency of fastapi, pinned here on purpose.
# fastapi's own range still admits starlette 0.48.x, which carries eight
# known CVEs (PYSEC-2026-161/248/249/1942/2280/2281), and pip will happily
# keep an already-installed vulnerable version that satisfies that range.
# Naming the floor directly is what actually forces the patched build.
# `make audit` checks this on every run.
starlette>=1.3.1,<2.0
uvicorn[standard]>=0.32,<0.36
python-multipart>=0.0.18,<0.1

# Machine learning
scikit-learn>=1.7.2,<1.8
joblib>=1.4,<2.0

# Data processing
pandas>=2.2,<3.0
numpy>=1.26,<3.0
openpyxl>=3.1,<4.0

# Validation and configuration
pydantic>=2.9,<3.0
pydantic-settings>=2.6,<3.0
python-dotenv>=1.0,<2.0
